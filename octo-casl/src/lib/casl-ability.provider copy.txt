import { Provider, Scope } from '@nestjs/common';
import { REQUEST } from '@nestjs/core';
import { Request } from 'express';
import { AbilityFactory } from './casl-ability.factory';
import { PrismaClient, User} from '@prisma/client';
import { AppAbility } from './types';
import { ABILITY_TOKEN,PRISMA_CLIENT } from './casl.constants'



export const AbilityProvider: Provider<AppAbility> = {
  provide: ABILITY_TOKEN,
  inject: [
    REQUEST,
    PRISMA_CLIENT
  ],
  useFactory: async (
    request: Request & { user?: User },
    prisma: PrismaClient  // Inject the PrismaClient
  ) => {
    const user = request.user;
    if (!user) {
      console.warn('No user found on request, creating guest ability');
      return new AbilityFactory(prisma).createForUser(1);
    }
    return new AbilityFactory(prisma).createForUser(user.roleId);
  },
  scope: Scope.REQUEST,
};
